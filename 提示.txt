# utils/signals.py
from PySide6.QtCore import QObject, Signal

class BackupSignals(QObject):
    # 参数：任务ID，当前文件进度(0-100)，文件名，总体进度(可选)
    task_progress = Signal(int, int, str)  
    # 参数：任务ID，成功标志，消息
    task_finished = Signal(int, bool, str) 
    # 参数：总体完成百分比
    overall_progress = Signal(int)          

# 创建全局单例
backup_bus = BackupSignals()
2. 定义具体的备份任务（BackupTask）
python
# tools/backup_task.py
from PySide6.QtCore import QRunnable, QThreadPool
from utils.signals import backup_bus
import zipfile
import os

class BackupTask(QRunnable):
    def __init__(self, task_id, file_path, dest_dir, zip_name):
        super().__init__()
        self.task_id = task_id
        self.file_path = file_path
        self.dest_dir = dest_dir
        self.zip_name = zip_name   # 例如 "project_2026.zip"
        self._is_cancelled = False  # 暂不支持中途取消单个任务，但留接口

    def run(self):
        """子线程中执行"""
        try:
            # 获取文件大小用于计算进度（细化到字节）
            total_size = os.path.getsize(self.file_path)
            processed_size = 0

            # 自定义进度回调（用于 zipfile 的写过程）
            def on_progress(written_bytes):
                nonlocal processed_size
                processed_size += written_bytes
                percent = int((processed_size / total_size) * 100)
                backup_bus.task_progress.emit(self.task_id, percent, os.path.basename(self.file_path))

            # 采用自定义文件包装器来捕获写入进度（复用之前讲的 ProgressFile 逻辑）
            dest_zip_path = os.path.join(self.dest_dir, self.zip_name)
            with zipfile.ZipFile(dest_zip_path, 'w', compression=zipfile.ZIP_DEFLATED) as zipf:
                # 这里为了简洁，先省略 ProgressFile 的详细实现，但原理相同
                # 实际这里会调用 zipf.write(...) 并触发回调
                zipf.write(self.file_path, arcname=os.path.basename(self.file_path))

            # 任务完成，发射完成信号
            backup_bus.task_finished.emit(self.task_id, True, f"{self.zip_name} 备份成功")

        except Exception as e:
            backup_bus.task_finished.emit(self.task_id, False, f"{self.zip_name} 失败: {str(e)}")
3. 在 UI 工具类中使用线程池

# tools/tool_backup.py
from PySide6.QtCore import QThreadPool
from PySide6.QtWidgets import QWidget, QVBoxLayout, QProgressBar, QPushButton, QLabel
from utils.signals import backup_bus
from tools.backup_task import BackupTask

class BackupToolWidget(QWidget):
    def __init__(self):
        super().__init__()
        # ... 初始化 UI ...
        self.total_progress = QProgressBar()
        self.status_label = QLabel("就绪")
        
        # 连接全局信号总线（在 UI 线程中接收）
        backup_bus.task_progress.connect(self.on_task_progress)
        backup_bus.task_finished.connect(self.on_task_finished)
        backup_bus.overall_progress.connect(self.total_progress.setValue)

        # 获取全局线程池实例
        self.thread_pool = QThreadPool.globalInstance()
        # 限制最大并发数（建议设为 CPU 核心数或 核心数*2，防止系统卡顿）
        self.thread_pool.setMaxThreadCount(4)  
        
        # 任务跟踪
        self.total_tasks = 0
        self.finished_tasks = 0

    def start_backup(self):
        """用户点击开始备份"""
        file_list = self.get_file_list()  # 假设返回 [(文件路径, 目标zip名), ...]
        self.total_tasks = len(file_list)
        self.finished_tasks = 0
        self.total_progress.setValue(0)

        # 遍历所有文件，将任务提交到线程池
        for idx, (file_path, zip_name) in enumerate(file_list):
            task = BackupTask(idx, file_path, self.des_dir_path, zip_name)
            # 提交任务（线程池会自动调度执行）
            self.thread_pool.start(task)

    def on_task_progress(self, task_id, percent, filename):
        """接收单个任务进度（运行在主线程）"""
        # 可以更新某个任务对应的进度条，这里简单打印或只更新状态标签
        self.status_label.setText(f"任务{task_id}: {filename} - {percent}%")

    def on_task_finished(self, task_id, success, msg):
        """单个任务完成回调"""
        self.finished_tasks += 1
        # 计算总体完成度
        overall = int(self.finished_tasks * 100 / self.total_tasks)
        backup_bus.overall_progress.emit(overall)  # 更新总进度条
        
        if not success:
            print(f"错误: {msg}")
        if self.finished_tasks == self.total_tasks:
            self.status_label.setText("全部备份完成！")
